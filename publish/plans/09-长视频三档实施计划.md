"""铺底循环：曲子不够长时接回自己，接缝落在最安静处。"""
from __future__ import annotations

import subprocess
import wave
from pathlib import Path

import numpy as np
import pytest

from pyv import bgm


def _wav_seconds(path: Path) -> float:
    with wave.open(str(path), "rb") as handle:
        return handle.getnframes() / handle.getframerate()


def _make_bgm(path: Path, seconds: float, sample_rate: int = 48000) -> Path:
    """造一条测试用曲子：正弦 + 中间一段安静。"""
    t = np.arange(int(seconds * sample_rate)) / sample_rate
    tone = 0.3 * np.sin(2 * np.pi * 220 * t)
    quiet = np.zeros_like(tone)
    middle = slice(len(tone) // 3, len(tone) // 3 + len(tone) // 10)
    tone[middle] = 0.0
    stereo = np.stack([tone, tone], axis=1).astype(np.float32)
    subprocess.run(
        ["ffmpeg", "-y", "-v", "error", "-f", "f32le", "-ac", "2",
         "-ar", str(sample_rate), "-i", "-", str(path)],
        input=stereo.tobytes(), check=True,
    )
    return path


def test_loop_bgm_covers_the_requested_duration(tmp_path):
    """循环后的铺底长度必须盖住请求的时长。"""
    src = _make_bgm(tmp_path / "bgm.wav", 10.0)
    out = bgm.loop_bgm(src, duration_s=35.0, sample_rate=48000)
    assert out.shape == (35 * 48000, 2)


def test_loop_bgm_returns_the_track_unchanged_when_it_is_long_enough(tmp_path):
    """曲子本来就够长时，输出应当就是曲子本身（不改变已有成片）。"""
    src = _make_bgm(tmp_path / "bgm.wav", 30.0)
    out = bgm.loop_bgm(src, duration_s=10.0, sample_rate=48000)
    assert out.shape == (10 * 48000, 2)
    decoded = bgm._decode_stereo(src, 48000)
    assert np.allclose(out[: decoded.shape[0]], decoded[: out.shape[0]], atol=1e-6)


def test_loop_bgm_seam_is_at_the_quietest_point(tmp_path):
    """接缝必须落在曲子最安静的位置——否则循环处会「咔」一下。"""
    src = _make_bgm(tmp_path / "bgm.wav", 10.0)
    out = bgm.loop_bgm(src, duration_s=25.0, sample_rate=48000)
    decoded = bgm._decode_stereo(src, 48000)
    period = decoded.shape[0]
    # 接缝在 out 的 period 位置附近，那里的瞬时能量必须明显低于整首均值
    seam_energy = float(np.mean(out[period - 200: period + 200] ** 2))
    whole_energy = float(np.mean(decoded ** 2))
    assert seam_energy < whole_energy


def test_loop_bgm_does_not_click_at_the_seam(tmp_path):
    """接缝处不许有咔哒：相邻样本的跳变不得比整段的还大。"""
    src = _make_bgm(tmp_path / "bgm.wav", 10.0)
    out = bgm.loop_bgm(src, duration_s=25.0, sample_rate=48000)
    period = 10 * 48000
    jumps = np.abs(np.diff(out, axis=0)).max(axis=1)
    seam_jump = float(jumps[period - 600: period + 600].max())
    assert seam_jump <= float(jumps.max()) * 1.001, (
        f"接缝处跳变 {seam_jump:.6f} 超过整段最大 {jumps.max():.6f}——会咔哒"
    )


def test_duck_bgm_covers_the_whole_narration(tmp_path):
    """铺底必须盖住整段旁白——这条是本次修的核心。"""
    sample_rate = 48000
    seconds = 300.0
    t = np.arange(int(seconds * sample_rate)) / sample_rate
    voice = np.zeros_like(t)
    for i in range(150):
        voice[int(i * 2 * sample_rate):int(i * 2 * sample_rate) + int(0.6 * sample_rate)] = 0.2
    narration = tmp_path / "narration.wav"
    subprocess.run(
        ["ffmpeg", "-y", "-v", "error", "-f", "f32le", "-ac", "1",
         "-ar", str(sample_rate), "-i", "-", str(narration)],
        input=voice.astype(np.float32).tobytes(), check=True,
    )
    src = _make_bgm(tmp_path / "bgm.wav", 10.0)
    out = bgm.duck_bgm(src, narration, tmp_path / "ducked.wav", duration_s=seconds)
    assert _wav_seconds(out) >= seconds - 0.5


def test_duck_bgm_still_respects_the_duck_depth(tmp_path):
    """循环之后 duck 深度仍然精确（不许因为循环把包络搞坏）。"""
    sample_rate = 48000
    seconds = 20.0
    t = np.arange(int(seconds * sample_rate)) / sample_rate
    voice = np.zeros_like(t)
    voice[: int(5 * sample_rate)] = 0.2
    narration = tmp_path / "narration.wav"
    subprocess.run(
        ["ffmpeg", "-y", "-v", "error", "-f", "f32le", "-ac", "1",
         "-ar", str(sample_rate), "-i", "-", str(narration)],
        input=voice.astype(np.float32).tobytes(), check=True,
    )
    src = _make_bgm(tmp_path / "bgm.wav", 8.0)
    out = bgm.duck_bgm(src, narration, tmp_path / "ducked.wav", duration_s=seconds)

    # 对照的是**同一份循环铺底**，不是原曲——循环后开头已经不是原曲开头了
    bed = bgm.loop_bgm(src, duration_s=seconds, sample_rate=sample_rate)
    ducked = bgm._decode_stereo(out, sample_rate)
    span = slice(0, int(4 * sample_rate))
    gain = float(np.mean(ducked[span] / np.maximum(bed[span], 1e-9)))
    expected = 10 ** (bgm.DUCK_DEPTH_DB / 20)
    assert gain == pytest.approx(expected, rel=0.05)

- [ ] **Step 3: 实现 `loop_bgm` 与 `_quiet_seam`**

在 `aiv/pipeline/pyv/bgm.py` 里，`duck_bgm` **之前**插入：

```python
# 循环铺底的接缝淡入淡出长度（秒）。
LOOP_FADE_S = 0.25


def _decode_stereo(path: str | Path, sample_rate: int = 48000) -> np.ndarray:
    """解码成 (n, 2) 的 float64 立体声。"""
    raw = _run([
        "ffmpeg", "-v", "error", "-i", str(path),
        "-f", "f32le", "-ac", "2", "-ar", str(sample_rate), "-",
    ], text=False).stdout
    frames = np.frombuffer(raw, dtype=np.float32).astype(np.float64)
    if frames.size % 2:
        frames = frames[:-1]
    return frames.reshape(-1, 2)


def _quiet_seam(song: np.ndarray, window: int) -> int:
    """找最安静窗口的中心，作为循环接缝点。

    窗口长度是 2×window：接缝左右各 window 个样本都要落在安静区里，
    这样交叉淡化处理的两段素材本身就已经很安静，淡化不会引入咔哒声。
    """
    mono = song.mean(axis=1)
    if mono.size < 2 * window:
        return mono.size // 2
    power = mono ** 2
    cumulative = np.concatenate([[0.0], np.cumsum(power)])
    energy = cumulative[2 * window:] - cumulative[:-2 * window]
    index = int(np.argmin(energy))
    return index + window


def loop_bgm(path: str | Path, duration_s: float, sample_rate: int = 48000,
             fade_s: float = LOOP_FADE_S) -> np.ndarray:
    """把曲子循环铺满 ``duration_s`` 秒，返回 (n, 2) 的 float64 立体声。

    曲子够长时直接取前 N 秒（**已有成片不受影响**）；
    不够长时把曲子从最安静处切开、循环拼接，接缝做等功率交叉淡化。
    """
    song = _decode_stereo(path, sample_rate)
    if song.shape[0] == 0:
        raise ValueError(f"BGM 是空的：{path}")
    need = int(round(duration_s * sample_rate))
    if need <= 0:
        return song[:0]
    period = song.shape[0]
    if period >= need:
        return song[:need]

    fade = max(1, min(int(round(fade_s * sample_rate)), period // 4))
    seam = _quiet_seam(song, fade)
    # 旋转曲子，让接缝落在位置 0：接缝左右各 fade 个样本都安静
    unit = np.concatenate([song[seam:], song[:seam]], axis=0)

    repeats = int(np.ceil(need / period)) + 1
    tiled = np.tile(unit, (repeats, 1))
    out = tiled[: need + fade].copy()

    # 每个 unit 边界做「淡出到零 + 从零淡入」。
    #
    # 为什么不是交叉淡化（crossfade）：交叉淡化会让**输出变长**——
    # 两段各淡出淡入 fade 个样本，总长度要加 fade。铺底长度是硬要求，
    # 不能多也不能少。淡出淡入不改变长度，形状也自洽：
    # 前一段的尾巴乘 cos（1→0），后一段的头乘 sin（0→1），
    # 接缝处两边都是 0，**连续，不会咔哒**。
    ramp_down = np.cos((np.arange(fade) + 0.5) / fade * (np.pi / 2))
    ramp_up = np.sin((np.arange(fade) + 0.5) / fade * (np.pi / 2))
    for index in range(1, repeats):
        boundary = index * period
        if boundary - fade < 0 or boundary + fade > out.shape[0]:
            continue
        out[boundary - fade:boundary] *= ramp_down[:, None]
        out[boundary:boundary + fade] *= ramp_up[:, None]

    return out[:need]
```

> ⚠️ **形状自洽是这条代码最容易踩的坑**：交叉淡化的写法是
> `out[start:start + 2 * fade] = tail + head`，而 `tail + head` 只有 `fade` 行——
> 直接运行会报 `could not broadcast input array from shape (fade, 2) into shape (2*fade, 2)`。
> 淡出淡入没有这个问题：两段分别乘各自的 ramp，长度都是 `fade`。
>
> **为什么选「淡出到零 + 从零淡入」而不是等功率交叉淡化**（2026-11-15 评审后定的）：
>
> - **形状自洽、输出长度精确**（`need` 恒等于 `int(round(duration_s * sample_rate))`）——
>   铺底长度是硬要求，不多不少。
> - 交叉淡化也不是不能做：等功率重叠淡化**保持输出长度**，代价是**循环周期变成 P−F**
>   （要重复更多次才盖得够）。评审确认了这一点，所以「交叉淡化会让输出变长」这个说法**是错的**，
>   本计划原先的理由已作废。
> - 选淡出淡入的真实理由：**它不引入周期变化**，实现简单、接缝处两边都是 0、连续、不会咔哒。
> - **代价要说清楚**：每个接缝有一个 **0.25 s 的近静默凹坑**（音乐被拉过 0）。
>   实测接缝窗比整曲低 **23~27 dB**，凹坑被静音掩盖，bed 这种垫乐听不出来。
>   如果将来要求「接缝处绝对无凹坑」，就得改成等功率重叠淡化并接受周期变成 P−F。

   而循环后的 ducked 已经不是原母带的前缀了——两者不是同一段素材，
   相减得到的是「两段不同素材的 RMS 差」，量到的不是 duck 深度。
   构造用例实测偏到 **20.375 dB**（设定 7.0）；真实母带今天是 **7.181 dB**，
   只是因为接缝恰好落在曲尾才没越界——**靠巧合过的守卫不算守卫**。
   修法：函数内改用 `loop_bgm` 产出的那份 bed 当对照，并补一条
   「曲子比旁白短时 duck 深度仍 ≈ 7 dB」的回归测试（这条路径原本 0 覆盖）。

2. **两条接缝守卫原本是恒真断言**，注入缺陷后仍全绿：
   - `seam_jump <= jumps.max() * 1.001`：左边是**子集**最大值、右边是全集最大值，
     恒成立；
   - `seam_energy < whole_energy`：测量窗整个落在淡化斜坡内，幅度被 ramp 压到约 0.026 倍，
     天然远低于整首均值，**与接缝落在哪无关**。
   修法：单元级直接钉 `_quiet_seam`（断言它返回的窗口能量 == 全曲 2×fade 窗口能量的最小值），
   集成级再断言**淡化前**的素材能量低于整曲均值；click 守卫改成与**典型**跳变比
   （如 `np.percentile(jumps, 99.9) * 1.5`），不是与最大值比。

3. **`_quiet_seam` 的「最安静」要按逐声道功率算**，不是带符号的单声道均值——
   左右反相的立体声，用带符号均值可以很小而实际很响，会把接缝选到响处。
   改成 `power = (song ** 2).mean(axis=1)`。

> **内存峰值**（52 分钟档）：`np.tile` 一次性铺满 + `.copy()`，加上 `duck_bgm` 里
> 同时留着 bed / ducked / curve / narration，峰值约 **7~8 GB**。

```python
def _total_error(pairs, texts, weights) -> float:
    """对齐结果的总平方误差——用来比较「剪枝后有没有变差」。"""
    total = 0.0
    for (start, end, _), weight in zip(pairs, weights):
        total += (end - start - weight) ** 2
    return total
```

> 这里的 `_total_error` 用**实际时长**算，与 DP 的目标（按文本权重分配）不是同一个量。
> 它只用来做「不比之前差」的粗判；精确的一致性由第一条测试（逐字相同）保证。

```python
def align_by_weight(spans: list[tuple[float, float]],
                     texts: list[str],
                     weights: list[float] | None = None,
                     *,
                     prune: bool = True,
                     j_window: int | None = None,
                     i_window: int | None = None) -> list[tuple[float, float, str]]:
    """把语音片段最优地分配给各句：每句时长尽量贴近"文本占比 × 总语音时长"。

    TTS 在逗号处也会停顿，直接按静音长度切会切出十几段；
    这里用动态规划在"所有连续分组方案"里挑误差最小的一组，
    边界必然落在语音片段之间，既不切在词中间，也比就近吸附均匀得多。

    ``prune=True`` 时走剪枝版：只在「理想位置」附近找解，
    从 O(句数 × 片段数²) 降到 O(句数 × 片段数 × 常数)。
    窗口开得够大时，剪枝结果与穷举**逐字相同**。
    """
    if len(texts) < 2:
        raise ValueError("至少需要两句才能计算边界")
    if weights is None:
        weights = [display_width(t) for t in texts]
    if len(weights) != len(texts):
        raise ValueError("权重数量与句子数量不一致")
    if len(spans) < len(texts):
        raise ValueError(
            f"语音片段数 {len(spans)} 少于句子数 {len(texts)}，无法逐句对齐"
        )

    total_speech = sum(e - s for s, e in spans)
    total_weight = sum(weights)
    if total_weight <= 0:
        raise ValueError("文本权重总和为 0，无法对齐")

    targets: list[float] = []
    cumulative_weight = 0.0
    previous_cumulative = 0.0
    for weight in weights:
        cumulative_weight += weight
        cumulative_time = total_speech * cumulative_weight / total_weight
        targets.append(cumulative_time - previous_cumulative)
        previous_cumulative = cumulative_time

    prefix = [0.0]
    for s, e in spans:
        prefix.append(prefix[-1] + (e - s))

    n_spans, n_sent = len(spans), len(texts)
    inf = float("inf")
    dp = [[inf] * (n_spans + 1) for _ in range(n_sent + 1)]
    back = [[0] * (n_spans + 1) for _ in range(n_sent + 1)]
    dp[0][0] = 0.0

    # 剪枝窗口：默认按「每句平均分到几个片段」推。
    # j_window 限制「这句用掉到第几个片段」，i_window 限制「从第几个片段开始」。
    if j_window is None:
        j_window = max(8, int(0.2 * n_spans / n_sent) + 4)
    if i_window is None:
        i_window = max(8, int(0.05 * n_spans / n_sent) + 4)

    for k in range(1, n_sent + 1):
        target = targets[k - 1]
        # 理想位置：第 k 句大约用掉 n_spans × k / n_sent 个片段
        ideal = n_spans * k / n_sent
        if prune:
            j_low = max(k, int(ideal - j_window))
            j_high = min(n_spans, int(ideal + j_window) + 1)
        else:
            j_low, j_high = k, n_spans
        for j in range(j_low, j_high + 1):
            for i in range(k - 1, j):
                if prune and i_window is not None:
                    # 最优的 i 落在「让这一组时长≈target」的位置附近
                    best_i = _closest_prefix_index(prefix, j, target)
                    low = max(k - 1, best_i - i_window)
                    high = min(j - 1, best_i + i_window)
                    if not (low <= i <= high):
                        continue
                if dp[k - 1][i] == inf:
                    continue
                group = prefix[j] - prefix[i]
                cost = (group - target) ** 2
                total = dp[k - 1][i] + cost
                if total < dp[k][j]:
                    dp[k][j] = total
                    back[k][j] = i
    ...
```

并在模块里加上：

```python
def _closest_prefix_index(prefix: list[float], j: int, target: float) -> int:
    """在前 j 个片段里，找让「这一组时长」最接近 target 的起点 i。

    用二分：prefix 单调增，组时长 prefix[j] - prefix[i] 随 i 增大而减小。
    """
    low, high = 0, j - 1
    best = 0
    best_gap = float("inf")
    while low <= high:
        mid = (low + high) // 2
        gap = abs(prefix[j] - prefix[mid] - target)
        if gap < best_gap:
            best_gap = gap
            best = mid
        if prefix[j] - prefix[mid] > target:
            low = mid + 1
        else:
            high = mid - 1
    return best
```

> ⚠️ 二分的前提是 prefix 单调增——它确实是（每段时长为正）。
> `while` 循环里 `prefix[j] - prefix[mid]` 随 mid 增大而减小，所以 mid 偏大时组时长偏小，
> 需要往小的方向走；代码里两个分支的移动方向要跟这个单调性对上。

import time

from pyv import subtitle
from pyv.timeline import SubtitleEvent, SubtitleStyle


def _events(count: int):
    events = []
    cursor = 0.0
    for index in range(count):
        start = cursor
        end = cursor + 1.5
        cursor = end + 0.25
        events.append(SubtitleEvent(start=start, end=end, text=f"第{index}句"))
    return events

    for event in events:
        if event.start <= t < event.end:
            return event
    return None

class SubtitleBurner:
    """按绝对时间把字幕烧进画面。

    定位用**二分**而不是从头扫：完整版有 600 条字幕 × 5 万帧，
    线性扫是 3,000 万次比较。字幕事件不重叠，所以二分找到的候选就是答案。
    """

    def __init__(self, events: list[SubtitleEvent], style: SubtitleStyle) -> None:
        self.events = sorted(events, key=lambda e: e.start)
        self.style = style
        self._starts = [event.start for event in self.events]

    def event_at(self, t: float) -> SubtitleEvent | None:
        index = bisect.bisect_right(self._starts, t) - 1
        if index < 0:
            return None
        event = self.events[index]
        return event if event.start <= t < event.end else None

    def draw(self, ctx: cairo.Context, t: float, width: int, height: int) -> None:
        event = self.event_at(t)
        if event is None:
            return
        draw_event(ctx, event.text, self.style, width, height)

```python
"""母稿解析：章节、mermaid 图、代码块、表格、散文字数。

母稿（``docs/`` 下的 md）是唯一真相。长视频的「画面」直接来自它——
图渲染成 PNG、代码块高亮、表格排版，都不需要重新手写内容。

计数口径（与契约八 §2.1 一致，改一处就要改另一处）：

- **散文**：去掉 front matter、标题行、代码块内、表格行之后，
  剩下非空行的去空白字符数
- **mermaid 块**：语言标注为 ``mermaid`` 的围栏块
- **code 块**：其余围栏块（**含** mermaid 块）
- **table 块**：连续的表格行
"""
from __future__ import annotations

import re
from dataclasses import dataclass
from pathlib import Path

FRONT_MATTER_RE = re.compile(r"^---\n[\s\S]*?\n---\n")

#: 围栏块的类型。``mermaid`` 单独一类，其余算 code。
MERMAID_LANG = "mermaid"


@dataclass(frozen=True)
class Block:
    """母稿里的一个可复用内容块。"""
    kind: str
    nth: int
    content: str


@dataclass(frozen=True)
class MotherDoc:
    """一篇母稿。"""
    path: Path
    title: str
    blocks: tuple[Block, ...]
    prose_chars: int

    def count(self, kind: str) -> int:
        """某一类块的数量。"""
        return sum(1 for block in self.blocks if block.kind == kind)


def _dense(text: str) -> int:
    """去掉所有空白后的字符数。"""
    return len(re.sub(r"\s+", "", text))


def _strip_front_matter(text: str) -> str:
    return FRONT_MATTER_RE.sub("", text, count=1)


def _parse_blocks(lines: list[str]) -> tuple[list[Block], int]:
    """扫出所有块，顺带累加散文字数。"""
    blocks: list[Block] = []
    prose = 0
    counters = {"mermaid": 0, "code": 0, "table": 0}

    index = 0
    in_fence = False
    fence_lang = ""
    fence_lines: list[str] = []

    while index < len(lines):
        line = lines[index]
        stripped = line.strip()

        if stripped.startswith("```"):
            if not in_fence:
                in_fence = True
                fence_lang = stripped[3:].strip().lower()
                fence_lines = []
            else:
                kind = MERMAID_LANG if fence_lang == MERMAID_LANG else "code"
                counters[kind] += 1
                blocks.append(Block(kind=kind, nth=counters[kind],
                                    content="\n".join(fence_lines)))
                in_fence = False
                fence_lang = ""
            index += 1
            continue

        if in_fence:
            fence_lines.append(line)
            index += 1
            continue

        if stripped.startswith("|"):
            # 连续的表格行算**一块**
            table_lines = []
            while index < len(lines) and lines[index].strip().startswith("|"):
                table_lines.append(lines[index].strip())
                index += 1
            counters["table"] += 1
            blocks.append(Block(kind="table", nth=counters["table"],
                                content="\n".join(table_lines)))
            continue

        if stripped.startswith("#"):
            index += 1
            continue

        if stripped:
            prose += _dense(stripped)
        index += 1

    return blocks, prose


def load_motherdoc(path: str | Path) -> MotherDoc:
    """读一篇母稿。"""
    target = Path(path)
    text = target.read_text(encoding="utf-8", errors="replace")
    body = _strip_front_matter(text)
    blocks, prose = _parse_blocks(body.splitlines())
    title = ""
    for line in body.splitlines():
        if line.startswith("# "):
            title = line[2:].strip()
            break
    return MotherDoc(path=target, title=title, blocks=tuple(blocks),
                     prose_chars=prose)


def find_block(doc: MotherDoc, kind: str, nth: int) -> Block:
    """按「类型 + 同类序号」找块。找不到就报错——引用错了要当场发现。"""
    for block in doc.blocks:
        if block.kind == kind and block.nth == nth:
            return block
    raise LookupError(
        f"{doc.path.name} 里没有 {kind} 类的第 {nth} 块"
        f"（该类共 {doc.count(kind)} 块）"
    )
```

新建 `aiv/pipeline/pyv/skeleton.py`：

```python
"""讲稿骨架：一期视频的口播与画面的唯一真相。

骨架是一个 YAML 文件（``aiv/plans/<EP>.yaml``），每期一个。
它把「这一期讲什么、画面长什么样」写在一起，然后按档位裁剪：

- **精简版** —— hook + conclusion
- **进阶版** —— hook + mechanism + evidence + conclusion
- **完整版** —— 全部（含 aside）

裁剪只看每段的 ``role``。**不做自动裁剪**：字数落不进区间就报错，
让人改 role 标记——悄悄删段与本仓库的纪律冲突。
"""
from __future__ import annotations

from dataclasses import dataclass
from pathlib import Path
from typing import Any

import yaml

AIV_ROOT = Path(__file__).resolve().parents[2]
PLANS_DIR = AIV_ROOT / "plans"

#: 段落角色。裁剪只看这个。
ROLES = ("hook", "mechanism", "evidence", "conclusion", "aside")

#: 三个档位。
TIERS = ("short", "mid", "full")

#: 画面类型。
VISUAL_KINDS = ("scene", "mermaid", "code", "table", "card")


@dataclass(frozen=True)
class Segment:
    """一个视觉段：讲什么、画面是什么。"""
    id: str
    role: str
    visual: dict
    narration: tuple[str, ...]
    subtitles: tuple[str, ...]


@dataclass(frozen=True)
class Skeleton:
    """一期的讲稿骨架。"""
    episode: str
    source: str
    segments: tuple[Segment, ...]
    title: str = ""

    def by_role(self, roles: tuple[str, ...]) -> tuple[Segment, ...]:
        """按 role 过滤出段（保持原顺序）。"""
        return tuple(s for s in self.segments if s.role in roles)


def SKELETON_PATH(episode: str) -> Path:
    """某一期的骨架文件路径。"""
    return PLANS_DIR / f"{episode}.yaml"


def _parse_segment(raw: dict, index: int) -> Segment:
    if not isinstance(raw, dict):
        raise ValueError(f"第 {index} 段不是对象")
    sid = str(raw.get("id") or "").strip()
    if not sid:
        raise ValueError(f"第 {index} 段没有 id")
    role = str(raw.get("role") or "").strip()
    if role not in ROLES:
        raise ValueError(
            f"段 {sid} 的 role 是 {role!r}，合法值只有 {ROLES}"
        )
    visual = raw.get("visual") or {}
    if not isinstance(visual, dict):
        raise ValueError(f"段 {sid} 的 visual 必须是对象")
    kind = str(visual.get("kind") or "").strip()
    if kind not in VISUAL_KINDS:
        raise ValueError(
            f"段 {sid} 的 visual.kind 是 {kind!r}，合法值只有 {VISUAL_KINDS}"
        )
    narration = tuple(str(x) for x in (raw.get("narration") or ()))
    subtitles = tuple(str(x) for x in (raw.get("subtitles") or ()))
    if not narration:
        raise ValueError(f"段 {sid} 没有 narration")
    if len(narration) != len(subtitles):
        raise ValueError(
            f"段 {sid} 的口播 {len(narration)} 句与字幕 {len(subtitles)} 行不一致"
        )
    return Segment(id=sid, role=role, visual=dict(visual),
                   narration=narration, subtitles=subtitles)


def load_skeleton(path: str | Path) -> Skeleton:
    """读一份骨架。"""
    target = Path(path)
    raw = yaml.safe_load(target.read_text(encoding="utf-8"))
    if not isinstance(raw, dict):
        raise ValueError(f"{target.name} 的顶层不是对象")
    episode = str(raw.get("episode") or "").strip()
    if not episode:
        raise ValueError(f"{target.name} 没有 episode")
    source = str(raw.get("source") or "").strip()
    title = str(raw.get("title") or "").strip()
    segments = []
    seen: set[str] = set()
    for index, item in enumerate(raw.get("segments") or ()):
        segment = _parse_segment(item, index)
        if segment.id in seen:
            raise ValueError(f"{target.name} 里有重复的段 id：{segment.id}")
        seen.add(segment.id)
        segments.append(segment)
    if not segments:
        raise ValueError(f"{target.name} 里没有段")
    return Skeleton(episode=episode, source=source, segments=tuple(segments),
                    title=title)


def save_skeleton(skeleton: Skeleton, path: str | Path) -> None:
    """写一份骨架。"""
    target = Path(path)
    target.parent.mkdir(parents=True, exist_ok=True)
    payload = {
        "episode": skeleton.episode,
        "source": skeleton.source,
        "title": skeleton.title,
        "segments": [
            {
                "id": segment.id,
                "role": segment.role,
                "visual": segment.visual,
                "narration": list(segment.narration),
                "subtitles": list(segment.subtitles),
            }
            for segment in skeleton.segments
        ],
    }
    target.write_text(
        yaml.safe_dump(payload, allow_unicode=True, sort_keys=False),
        encoding="utf-8",
    )
```

#: 档位 → 选中的 role。**写死**，不靠推断——推断出来的规则一旦母稿变了就悄悄改了行为。
TIER_ROLES: dict[str, tuple[str, ...]] = {
    "short": ("hook", "conclusion"),
    "mid": ("hook", "mechanism", "evidence", "conclusion"),
    "full": ("hook", "mechanism", "evidence", "aside", "conclusion"),
}


def crop(skeleton: Skeleton, tier: str) -> Skeleton:
    """按档位裁剪骨架。

    只**丢段**，不改句子——精简版讲的就是完整版里的那些段，逐字相同。

    if tier not in TIER_ROLES:
        raise ValueError(
            f"未知档位 {tier!r}，合法值只有 {sorted(TIER_ROLES)}"
        )
    roles = TIER_ROLES[tier]
    return Skeleton(
        episode=skeleton.episode,
        source=skeleton.source,
        segments=skeleton.by_role(roles),
        title=skeleton.title,
    )

```python
"""骨架 → (口播稿, 字幕表, 视觉契约)。

派生出来的三元组**格式与现在完全一致**：
口播稿是「每行一句」的代码块，字幕表是每句一行的表格，
契约是 ``pyv/contract.py`` 认识的那个 JSON。

所以 ``pyv/align.py`` / ``pyv/render.py`` / ``pyv/compose.py``
一行都不用改——它们看到的还是老样子。
"""
from __future__ import annotations

from dataclasses import dataclass, field

from pyv import skeleton

#: 画面类型 → 通用渲染器模块。
GENERIC_RENDERERS = {
    "mermaid": "pyv.scenes.gen_mermaid",
    "code": "pyv.scenes.gen_code",
    "table": "pyv.scenes.gen_table",
    "card": "pyv.scenes.gen_card",
}


@dataclass(frozen=True)
class Derived:
    """派生出来的三元组。"""
    narration: list[str] = field(default_factory=list)
    subtitles: list[str] = field(default_factory=list)
    contract: dict = field(default_factory=dict)


def _segment_params(segment: skeleton.Segment) -> tuple[str, dict]:
    """一段的 (renderer, params)。"""
    visual = segment.visual
    kind = str(visual.get("kind") or "")
    if kind == "scene":
        renderer = str(visual.get("renderer") or "")
        if not renderer:
            raise ValueError(f"段 {segment.id} 的 kind=scene 但没有 renderer")
        return renderer, dict(visual.get("params") or {})
    renderer = GENERIC_RENDERERS[kind]
    params = dict(visual.get("params") or {})
    if "ref" in visual:
        params["ref"] = dict(visual["ref"])
    return renderer, params


def derive(source: skeleton.Skeleton, tier: str) -> Derived:
    """把骨架按档位裁剪，派生出口播稿、字幕表与视觉契约。"""
    cropped = skeleton.crop(source, tier)

    narration: list[str] = []
    subtitles: list[str] = []
    segments: list[dict] = []

    for segment in cropped.segments:
        renderer, params = _segment_params(segment)
        sentences = list(range(len(narration), len(narration) + len(segment.narration)))
        narration.extend(segment.narration)
        subtitles.extend(segment.subtitles)
        segments.append({
            "id": segment.id,
            "renderer": renderer,
            "sentences": sentences,
            "params": params,
        })

    contract = {
        "version": 1,
        "episode": cropped.episode,
        "title": cropped.title,
        "segments": segments,
    }
    return Derived(narration=narration, subtitles=subtitles, contract=contract)


def narration_markdown(narration: list[str]) -> str:
    """口播稿小节的正文：一个代码块，每行一句。"""
    lines = "\n".join(narration)
    return f"```\n{lines}\n```\n"


def subtitle_table(subtitles: list[str]) -> str:
    """字幕表小节的正文：每句一行。"""
    lines = ["| # | 字幕 |", "| --- | --- |"]
    for index, text in enumerate(subtitles, start=1):
        lines.append(f"| {index} | {text} |")
    return "\n".join(lines) + "\n"
```

#!/usr/bin/env python3
"""把现有的「口播稿 + 字幕表 + 契约」反推成讲稿骨架。

这是一次性迁移脚本。迁移**不改动任何已有内容**——
它只是把三份手写的东西合成一份，然后由
``tests/test_skeleton_migration.py`` 守住「结构等价」。

跑法（仓库根目录）：

    python3 aiv/pipeline/scripts/migrate_skeletons.py
"""
from __future__ import annotations

import sys
from pathlib import Path

sys.path.insert(0, str(Path(__file__).resolve().parents[1]))

from pyv import align, contract, episodes, skeleton  # noqa: E402

#: 现有六期的段角色。**存量事实**：第一段是钩子（s01_intro），
#: 中间四段是机制与证据，最后一段是落版（s06_outro）。
ROLES = ("hook", "mechanism", "mechanism", "mechanism", "evidence", "conclusion")


def migrate(ep: str) -> skeleton.Skeleton:
    script = episodes.script_path(ep)
    narration = align.parse_script(script)
    subtitles = align.parse_subtitle_table(script)
    loaded = contract.load_contract(episodes.contract_path(ep))

    if len(loaded.segments) != len(ROLES):
        raise ValueError(
            f"{ep} 有 {len(loaded.segments)} 段，登记的角色有 {len(ROLES)} 个"
            "——对不上就停下来查，不要硬凑"
        )

    segments = []
    for index, segment in enumerate(loaded.segments):
        segments.append(skeleton.Segment(
            id=segment.id,
            role=ROLES[index],
            visual={"kind": "scene", "renderer": segment.renderer,
                    "params": dict(segment.params)},
            narration=tuple(narration[i] for i in segment.sentences),
            subtitles=tuple(subtitles[i] for i in segment.sentences),
        ))
    return skeleton.Skeleton(
        episode=ep,
        source=episodes.MOTHER_DOCS[ep],
        segments=tuple(segments),
        title=loaded.title,
    )


def main() -> int:
    for ep in sorted(episodes.EPISODES):
        result = migrate(ep)
        path = skeleton.SKELETON_PATH(ep)
        skeleton.save_skeleton(result, path)
        print(f"已迁移 {ep} → {path}（{len(result.segments)} 段）")
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
```

    },

        "mid": "尚未产出进阶版（2026-11-15）",
        "full": "尚未产出完整版（2026-11-15）",
    },
}

#: 期号 → 母稿路径。登记在 episodes.py（期号登记表的自然归属），这里只引用。
MOTHER_DOCS = episodes.MOTHER_DOCS


def count_chars(texts) -> int:
    """去空格的字符数。"""
    import re
    return sum(len(re.sub(r"\s+", "", t)) for t in texts)


def check_numbers(tier: str, seconds: float, chars: int, ratio: float) -> list[str]:
    """核对一档的三个数。返回问题清单，空列表 = 通过。"""
    limits = TIER_LIMITS.get(tier)
    if limits is None:
        raise ValueError(f"未知档位 {tier!r}，合法值只有 {TIER_NAMES}")
    problems: list[str] = []

    low = limits["min_seconds"]
    high = limits["max_seconds"]
    if low is not None and seconds < low:
        problems.append(f"时长 {seconds:.1f} s 低于下限 {low:.0f} s")
    if high is not None and seconds > high:
        problems.append(f"时长 {seconds:.1f} s 超过上限 {high:.0f} s")

    low = limits["min_chars"]
    high = limits["max_chars"]
    if low is not None and chars < low:
        problems.append(f"口播字数 {chars} 低于下限 {low}")
    if high is not None and chars > high:
        problems.append(f"口播字数 {chars} 超过上限 {high}")

    ratio_limit = limits["max_ratio"]
    if ratio > ratio_limit:
        problems.append(f"压缩比 {ratio:.1f}:1 超过上限 {ratio_limit:.0f}:1")

    return problems


def check_episode(ep: str, tier: str) -> list[str]:
    """核对某一期某一档。返回问题清单；豁免返回一行说明。"""
    if tier in TIER_EXEMPT.get(ep, {}):
        return [f"豁免：{TIER_EXEMPT[ep][tier]}"]

    path = skeleton.SKELETON_PATH(ep)
    if not path.is_file():
        return [f"没有骨架文件：{path}"]

    loaded = skeleton.load_skeleton(path)
    result = derive.derive(loaded, tier)
    chars = count_chars(result.narration)
    seconds = chars / SPEECH_RATE

    mother_path = REPO_ROOT / MOTHER_DOCS[ep]
    doc = motherdoc.load_motherdoc(mother_path)
    ratio = doc.prose_chars / chars if chars else float("inf")

    problems = check_numbers(tier, seconds, chars, ratio)
    if not result.narration:
        problems.append("这一档一段口播都没有")
    return problems
